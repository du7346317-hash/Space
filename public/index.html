const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const nodemailer = require('nodemailer'); const crypto = require('crypto'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: '*' } }); app.use(express.json()); // Transporter do Nodemailer para envio de e-mails de verificação const transporter = nodemailer.createTransport({ service: 'gmail', // ou o provedor SMTP de sua escolha auth: { user: process.env.EMAIL_USER, pass: process.env.EMAIL_PASS } }); // Rota de Cadastro com envio de confirmação app.post('/api/auth/register', async (req, res) => { const { username, email, password } = req.body; const token = crypto.randomBytes(32).toString('hex'); // 1. Salvar usuário no BD com is_verified = false e token de verificação // 2. Enviar e-mail de confirmação const verifyLink = `http://localhost:3000/api/auth/verify?token=${token}`; await transporter.sendMail({ from: '"Sua Plataforma" ', to: email, subject: 'Confirme seu e-mail para ativar a conta', html: `

Olá ${username}, clique no link para ativar sua conta:

${verifyLink}` }); res.json({ message: 'E-mail de verificação enviado! Verifique sua caixa de entrada.' }); }); // Socket.io: Gerenciamento em Tempo Real & WebRTC Voice Indicator io.on('connection', (socket) => { console.log('Usuário conectado:', socket.id); // Entrar em um canal de voz socket.on('join-voice-channel', ({ channelId, userId }) => { socket.join(`voice-${channelId}`); socket.to(`voice-${channelId}`).emit('user-connected', userId); }); // Indicador de Voz (Borda Verde) socket.on('speaking-status', ({ userId, isSpeaking, channelId }) => { io.to(`voice-${channelId}`).emit('user-speaking', { userId, isSpeaking }); }); // Sinalização WebRTC (Oferta, Resposta, Candidatos ICE) socket.on('webrtc-signal', ({ to, signal, from }) => { io.to(to).emit('webrtc-signal', { signal, from }); }); }); server.listen(3000, () => console.log('Servidor rodando na porta 3000'));
